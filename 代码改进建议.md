```python
# utils/performance.py
import time
import functools
from concurrent.futures import ThreadPoolExecutor, as_completed

def timing_decorator(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start_time = time.time()
        result = func(*args, **kwargs)
        end_time = time.time()
        print(f"{func.__name__} 执行时间: {end_time - start_time:.2f}秒")
        return result
    return wrapper

class BatchConverter:
    def __init__(self, max_workers=4):
        self.max_workers = max_workers
    
    def convert_files_parallel(self, file_list, output_format):
        """并行转换多个文件"""
        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = {
                executor.submit(self.convert_single_file, file_path, output_format): file_path 
                for file_path in file_list
            }
            
            results = []
            for future in as_completed(futures):
                file_path = futures[future]
                try:
                    result = future.result()
                    results.append(result)
                except Exception as e:
                    print(f"转换失败 {file_path}: {e}")
            
            return results
```

## 🔒 安全性改进

### 1. 输入验证

```python
# utils/validators.py
import os
from pathlib import Path

class FileValidator:
    def __init__(self, max_size=100*1024*1024):  # 100MB
        self.max_size = max_size
        self.allowed_extensions = {'.md', '.docx', '.html', '.pdf', '.epub'}
    
    def validate_file(self, file_path):
        path = Path(file_path)
        
        # 检查文件是否存在
        if not path.exists():
            raise ValueError(f"文件不存在: {file_path}")
        
        # 检查文件大小
        if path.stat().st_size > self.max_size:
            raise ValueError(f"文件过大: {file_path}")
        
        # 检查文件扩展名
        if path.suffix.lower() not in self.allowed_extensions:
            raise ValueError(f"不支持的文件格式: {path.suffix}")
        
        # 检查文件路径安全性
        if '..' in str(path) or path.is_absolute() and not str(path).startswith('/Users'):
            raise ValueError(f"不安全的文件路径: {file_path}")
        
        return True
```

### 2. 临时文件管理

```python
# utils/temp_manager.py
import tempfile
import atexit
import shutil
from pathlib import Path

class TempFileManager:
    def __init__(self):
        self.temp_dirs = []
        atexit.register(self.cleanup)
    
    def create_temp_dir(self):
        temp_dir = tempfile.mkdtemp(prefix='docuflow_')
        self.temp_dirs.append(temp_dir)
        return Path(temp_dir)
    
    def cleanup(self):
        for temp_dir in self.temp_dirs:
            try:
                shutil.rmtree(temp_dir)
            except Exception as e:
                print(f"清理临时目录失败: {temp_dir}, 错误: {e}")
```

```python
# utils/analytics.py
import json
from datetime import datetime
from pathlib import Path

class UsageAnalytics:
    def __init__(self):
        self.stats_file = Path.home() / '.docuflow' / 'stats.json'
        self.stats_file.parent.mkdir(exist_ok=True)
    
    def record_conversion(self, input_format, output_format, success=True):
        stats = self.load_stats()
        
        today = datetime.now().strftime('%Y-%m-%d')
        if today not in stats:
            stats[today] = {'conversions': [], 'success_count': 0, 'error_count': 0}
        
        stats[today]['conversions'].append({
            'input_format': input_format,
            'output_format': output_format,
            'timestamp': datetime.now().isoformat(),
            'success': success
        })
        
        if success:
            stats[today]['success_count'] += 1
        else:
            stats[today]['error_count'] += 1
        
        self.save_stats(stats)
    
    def load_stats(self):
        if self.stats_file.exists():
            return json.loads(self.stats_file.read_text())
        return {}
    
    def save_stats(self, stats):
        self.stats_file.write_text(json.dumps(stats, indent=2))
```
