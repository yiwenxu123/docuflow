### 4. 性能优化

#### 4.1 异步处理
```python
import asyncio
from concurrent.futures import ThreadPoolExecutor

class AsyncDocumentConverter:
    def __init__(self, max_workers: int = 4):
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    async def convert_file_async(self, input_path: str, output_format: str) -> ConversionResult:
        loop = asyncio.get_event_loop()
        return await loop.run_in_executor(
            self.executor, 
            self._convert_sync, 
            input_path, 
            output_format
        )
    
    async def convert_batch_async(self, file_list: List[Tuple[str, str]]) -> List[ConversionResult]:
        tasks = [self.convert_file_async(input_path, output_format) 
                for input_path, output_format in file_list]
        return await asyncio.gather(*tasks)
```

#### 4.2 缓存机制
```python
from functools import lru_cache
import hashlib

class CachedConverter:
    def __init__(self, cache_dir: str = ".docuflow_cache"):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(exist_ok=True)
    
    def _get_file_hash(self, file_path: str) -> str:
        """计算文件哈希值"""
        with open(file_path, 'rb') as f:
            return hashlib.md5(f.read()).hexdigest()
    
    def convert_with_cache(self, input_path: str, output_format: str) -> str:
        file_hash = self._get_file_hash(input_path)
        cache_key = f"{file_hash}_{output_format}"
        cache_file = self.cache_dir / f"{cache_key}.md"
        
        if cache_file.exists():
            return str(cache_file)
        
        # 执行转换并缓存结果
        result = self._convert_file(input_path, output_format)
        shutil.copy2(result, cache_file)
        return result
```

```python
from typing import Callable, Optional

class ProgressCallback:
    def __init__(self, callback: Optional[Callable[[int, str], None]] = None):
        self.callback = callback
    
    def update(self, progress: int, message: str = ""):
        if self.callback:
            self.callback(progress, message)

class DocumentConverter:
    def convert_file(self, input_path: str, output_format: str, 
                    progress_callback: Optional[ProgressCallback] = None) -> ConversionResult:
        if progress_callback:
            progress_callback.update(0, "开始转换...")
        
        # 转换逻辑
        if progress_callback:
            progress_callback.update(50, "处理中...")
        
        # 完成转换
        if progress_callback:
            progress_callback.update(100, "转换完成")
```

```python
from PySide6.QtWidgets import QWidget
from PySide6.QtCore import Signal

class DragDropWidget(QWidget):
    files_dropped = Signal(list)
    
    def __init__(self):
        super().__init__()
        self.setAcceptDrops(True)
    
    def dragEnterEvent(self, event):
        if event.mimeData().hasUrls():
            event.acceptProposedAction()
    
    def dropEvent(self, event):
        files = [url.toLocalFile() for url in event.mimeData().urls()]
        self.files_dropped.emit(files)
```

### 7. 监控和日志

#### 7.1 结构化日志
```python
import structlog

logger = structlog.get_logger()

class DocumentConverter:
    def convert_file(self, input_path: str, output_format: str) -> ConversionResult:
        logger.info(
            "conversion_started",
            input_path=input_path,
            output_format=output_format,
            file_size=Path(input_path).stat().st_size
        )
        
        try:
            # 转换逻辑
            result = self._do_conversion(input_path, output_format)
            
            logger.info(
                "conversion_completed",
                input_path=input_path,
                output_path=result.output_path,
                duration=result.duration,
                success=result.success
            )
            
            return result
        except Exception as e:
            logger.error(
                "conversion_failed",
                input_path=input_path,
                error=str(e),
                error_type=type(e).__name__
            )
            raise
```

#### 7.2 性能监控
```python
import time
from functools import wraps

def monitor_performance(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        start_time = time.time()
        try:
            result = func(*args, **kwargs)
            duration = time.time() - start_time
            logger.info(
                "performance_metric",
                function=func.__name__,
                duration=duration,
                success=True
            )
            return result
        except Exception as e:
            duration = time.time() - start_time
            logger.error(
                "performance_metric",
                function=func.__name__,
                duration=duration,
                success=False,
                error=str(e)
            )
            raise
    return wrapper
```

```python
app = FastAPI(title="DocuFlow API")

@app.post("/convert")
async def convert_document(file: UploadFile = File(...), output_format: str = "md"):
    # 保存上传文件
```
